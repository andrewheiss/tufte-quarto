% ----------------------
% Margin content stuff
% ----------------------
% Paragraph spacing in margin content is goofy because there are two types of
% margin content in Quarto:
%
% .column-margin divs:
%
% ::: {.column-margin}
% Here's some text.
%
% More text here.
% :::
%
% ...and footnotes:
%
% [^note]: Here's a note.
%
%     More text here.
%
%
% .column-margin divs use marginnote; footnotes use sidenotes. 
% Both packages work differently :(

% Add paragraph spacing for marginnotes
\AtBeginDocument{%
  \renewcommand{\marginfont}{\setlength{\parskip}{5pt}\setlength{\parindent}{0pt}}
}

% Add paragraph spacing for sidenotes
\makeatletter
\let\oldmarginparreset\@marginparreset
\renewcommand{\@marginparreset}{%
  \oldmarginparreset
  \setlength{\parskip}{5pt}%
  \setlength{\parindent}{0pt}%
}
\makeatother

% Recreate Quarto/Pandoc's CSL reference list but shrink the space between items
% to be 0.35 × baseline
\AtBeginDocument{%
  \renewenvironment{CSLReferences}[2]
  {\begin{list}{}{%
    \setlength{\itemindent}{0pt}%
    \setlength{\leftmargin}{0pt}%
    \setlength{\parsep}{0pt}%
    \ifodd #1
    \setlength{\leftmargin}{\cslhangindent}%
    \setlength{\itemindent}{-\cslhangindent}%
    \fi
    \setlength{\itemsep}{0.35\baselineskip}}}  % This is different from original
  {\end{list}}%
}
